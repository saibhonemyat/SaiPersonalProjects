#!/bin/bash
mkdir -p results
for level in levels/0_i_can_cook/*.yaml levels/1_we_can_cook/*.yaml levels/2_overcooked/*.yaml; do
  name=$(basename "$level" .yaml)
  echo "Running $name"
  uv run cook agent --level "$level" --headless --time-limit 180 \
    --result-out-file "results/$name.json" > "results/$name.log" 2>&1
done
uv run python - << 'EOF'
import json, glob, os
for path in sorted(glob.glob("results/*.json")):
    r = json.load(open(path))
    print(f"{os.path.basename(path)[:-5]:50} {r.get('reason'):10} score={r.get('score'):4} "
          f"delivered={r.get('orders_delivered')} left={r.get('orders_remaining')} steps={r.get('timesteps')}")
EOF
SCRIPT
chmod +x run_levels.sh
